FRP_PIPELINE PACKAGE BODY   36   BEGIN
                                                                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY   37     -- Pull the raw text we already cached during ingest

FRP_PIPELINE PACKAGE BODY   39      WHERE doc_id = FRP_INGEST.stable_doc_id(p_object_name);
                                                                                                                                                                                         
FRP_PIPELINE PACKAGE BODY   40 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY   41     l_system :=

FRP_PIPELINE PACKAGE BODY   69   -- ─────────────────────────────────────────────────────────
                                                                                                                                                                                       
FRP_PIPELINE PACKAGE BODY   70   -- match_requirements · vector search per requirement
                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY   71   -- ─────────────────────────────────────────────────────────
                                                                                                                                                                                       
FRP_PIPELINE PACKAGE BODY   72   FUNCTION match_requirements(p_parsed_json IN CLOB, p_top_k IN NUMBER DEFAULT 5)
                                                                                                                                                                    
FRP_PIPELINE PACKAGE BODY   73     RETURN CLOB IS
                                                                                                                                                                                                                                   
FRP_PIPELINE PACKAGE BODY   74     l_out      CLOB;
                                                                                                                                                                                                                                 
FRP_PIPELINE PACKAGE BODY   75     l_qvec     VECTOR;
                                                                                                                                                                                                                               
FRP_PIPELINE PACKAGE BODY   76     l_section  VARCHAR2(512);
                                                                                                                                                                                                                        
FRP_PIPELINE PACKAGE BODY   77     l_text     VARCHAR2(4000);
                                                                                                                                                                                                                       
FRP_PIPELINE PACKAGE BODY   78     l_first    BOOLEAN := TRUE;
                                                                                                                                                                                                                      
FRP_PIPELINE PACKAGE BODY   79     l_first_h  BOOLEAN;
                                                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY   80   BEGIN
                                                                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY   81     DBMS_LOB.CREATETEMPORARY(l_out, TRUE);
                                                                                                                                                                                                           
FRP_PIPELINE PACKAGE BODY   82     DBMS_LOB.WRITEAPPEND(l_out, 1, '[');
                                                                                                                                                                                                             
FRP_PIPELINE PACKAGE BODY   83 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY   84     FOR req IN (
                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY   85       SELECT
                                                                                                                                                                                                                                         
FRP_PIPELINE PACKAGE BODY   86         JSON_VALUE(jt.requirement, '$.section') AS section,
                                                                                                                                                                                          
FRP_PIPELINE PACKAGE BODY   87         JSON_VALUE(jt.requirement, '$.text')    AS req_text
                                                                                                                                                                                          
FRP_PIPELINE PACKAGE BODY   88       FROM JSON_TABLE(p_parsed_json, '$.requirements[*]'
                                                                                                                                                                                             
FRP_PIPELINE PACKAGE BODY   89         COLUMNS (requirement FORMAT JSON PATH '$')) jt
                                                                                                                                                                                               
FRP_PIPELINE PACKAGE BODY   90     ) LOOP
                                                                                                                                                                                                                                           
FRP_PIPELINE PACKAGE BODY   91       EXIT WHEN req.req_text IS NULL;
                                                                                                                                                                                                                
FRP_PIPELINE PACKAGE BODY   92 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY   93       -- Embed the requirement (Cohere Embed v4, query mode)
                                                                                                                                                                                         
FRP_PIPELINE PACKAGE BODY   94       l_qvec := DBMS_VECTOR_CHAIN.UTL_TO_EMBEDDING(
                                                                                                                                                                                                  
FRP_PIPELINE PACKAGE BODY   95         req.req_text,
                                                                                                                                                                                                                                
FRP_PIPELINE PACKAGE BODY   96         JSON('{
                                                                                                                                                                                                                                      
FRP_PIPELINE PACKAGE BODY   97           "provider": "ocigenai",
                                                                                                                                                                                                                    
FRP_PIPELINE PACKAGE BODY   98           "credential_name": "OCI$RESOURCE_PRINCIPAL",
                                                                                                                                                                                               
FRP_PIPELINE PACKAGE BODY   99           "url": "https://inference.generativeai.us-chicago-1.oci.oraclecloud.com/20231130/actions/embedText",
                                                                                                                                       
FRP_PIPELINE PACKAGE BODY  100           "model": "cohere.embed-v4.0",
                                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY  101           "input_type": "search_query"
                                                                                                                                                                                                               
FRP_PIPELINE PACKAGE BODY  102         }')
                                                                                                                                                                                                                                          
FRP_PIPELINE PACKAGE BODY  103       );
                                                                                                                                                                                                                                             
FRP_PIPELINE PACKAGE BODY  104 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  105       IF NOT l_first THEN DBMS_LOB.WRITEAPPEND(l_out, 1, ','); END IF;
                                                                                                                                                                               
FRP_PIPELINE PACKAGE BODY  106       l_first := FALSE;
                                                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY  107       DBMS_LOB.APPEND(l_out, '{"requirement":' ||
                                                                                                                                                                                                    
FRP_PIPELINE PACKAGE BODY  108         JSON_OBJECT('section' VALUE req.section, 'text' VALUE req.req_text) ||
                                                                                                                                                                       
FRP_PIPELINE PACKAGE BODY  109         ',"matches":[');
                                                                                                                                                                                                                             
FRP_PIPELINE PACKAGE BODY  110 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  111       l_first_h := TRUE;
                                                                                                                                                                                                                             
FRP_PIPELINE PACKAGE BODY  112       FOR h IN (
                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  113         SELECT c.chunk_text, d.filename, d.client_name,
                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY  114                VECTOR_DISTANCE(c.embedding, l_qvec, COSINE) AS dist
                                                                                                                                                                                  
FRP_PIPELINE PACKAGE BODY  115           FROM FRP_CHUNKS c JOIN FRP_DOCS d ON d.doc_id = c.doc_id
                                                                                                                                                                                   
FRP_PIPELINE PACKAGE BODY  116          WHERE d.deal_status = 'won'
                                                                                                                                                                                                                 
FRP_PIPELINE PACKAGE BODY  117          ORDER BY dist
                                                                                                                                                                                                                               
FRP_PIPELINE PACKAGE BODY  118          FETCH FIRST p_top_k ROWS ONLY
                                                                                                                                                                                                               
FRP_PIPELINE PACKAGE BODY  119       ) LOOP
                                                                                                                                                                                                                                         
FRP_PIPELINE PACKAGE BODY  120         IF NOT l_first_h THEN DBMS_LOB.WRITEAPPEND(l_out, 1, ','); END IF;
                                                                                                                                                                           
FRP_PIPELINE PACKAGE BODY  121         l_first_h := FALSE;
                                                                                                                                                                                                                          
FRP_PIPELINE PACKAGE BODY  122         DBMS_LOB.APPEND(l_out, JSON_OBJECT(
                                                                                                                                                                                                          
FRP_PIPELINE PACKAGE BODY  123           'filename'    VALUE h.filename,
                                                                                                                                                                                                            
FRP_PIPELINE PACKAGE BODY  124           'client_name' VALUE h.client_name,
                                                                                                                                                                                                         
FRP_PIPELINE PACKAGE BODY  125           'distance'    VALUE h.dist,
                                                                                                                                                                                                                
FRP_PIPELINE PACKAGE BODY  126           'chunk_text'  VALUE DBMS_LOB.SUBSTR(h.chunk_text, 1500, 1)
                                                                                                                                                                                 
FRP_PIPELINE PACKAGE BODY  127         ));
                                                                                                                                                                                                                                          
FRP_PIPELINE PACKAGE BODY  128       END LOOP;
                                                                                                                                                                                                                                      
FRP_PIPELINE PACKAGE BODY  129 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  130       DBMS_LOB.WRITEAPPEND(l_out, 2, ']}');
                                                                                                                                                                                                          
FRP_PIPELINE PACKAGE BODY  131     END LOOP;
                                                                                                                                                                                                                                        
FRP_PIPELINE PACKAGE BODY  132 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  133     DBMS_LOB.WRITEAPPEND(l_out, 1, ']');
                                                                                                                                                                                                             
FRP_PIPELINE PACKAGE BODY  134     RETURN l_out;
                                                                                                                                                                                                                                    
FRP_PIPELINE PACKAGE BODY  135   END match_requirements;
                                                                                                                                                                                                                            
FRP_PIPELINE PACKAGE BODY  136 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  137   -- ─────────────────────────────────────────────────────────
                                                                                                                                                                                       
FRP_PIPELINE PACKAGE BODY  138   -- draft_proposal · Grok-4 (long context)

FRP_PIPELINE PACKAGE BODY  142     l_system CLOB;
                                                                                                                                                                                                                                   
FRP_PIPELINE PACKAGE BODY  143     l_user   CLOB;
                                                                                                                                                                                                                                   
FRP_PIPELINE PACKAGE BODY  144   BEGIN
                                                                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY  145     l_system :=
                                                                                                                                                                                                                                      
FRP_PIPELINE PACKAGE BODY  146       'You are a senior proposal writer. Write a complete, persuasive ' ||

FRP_PIPELINE PACKAGE BODY  158 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  159   -- ─────────────────────────────────────────────────────────
                                                                                                                                                                                       
FRP_PIPELINE PACKAGE BODY  160   -- process_dropzone · ingest + parse + match + draft + persist
                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  161   -- ─────────────────────────────────────────────────────────
                                                                                                                                                                                       
FRP_PIPELINE PACKAGE BODY  162   FUNCTION process_dropzone(p_object_name IN VARCHAR2) RETURN VARCHAR2 IS
                                                                                                                                                                            
FRP_PIPELINE PACKAGE BODY  163     l_rfp_doc_id  VARCHAR2(64);
                                                                                                                                                                                                                      
FRP_PIPELINE PACKAGE BODY  164     l_proposal_id VARCHAR2(64) := LOWER(SYS_GUID());
                                                                                                                                                                                                 
FRP_PIPELINE PACKAGE BODY  165     l_parsed      CLOB;
                                                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY  166     l_matches     CLOB;
                                                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY  167     l_draft       CLOB;
                                                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY  168     l_client      VARCHAR2(256);
                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  169     l_due         VARCHAR2(20);
                                                                                                                                                                                                                      
FRP_PIPELINE PACKAGE BODY  170   BEGIN
                                                                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY  171     -- 1. Ingest the RFP itself so we have raw_text + a doc row
                                                                                                                                                                                      
FRP_PIPELINE PACKAGE BODY  172     FRP_INGEST.ingest_object(p_object_name);
                                                                                                                                                                                                         
FRP_PIPELINE PACKAGE BODY  173     l_rfp_doc_id := FRP_INGEST.stable_doc_id(p_object_name);
                                                                                                                                                                                         
FRP_PIPELINE PACKAGE BODY  174 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  175     -- 2. Parse + match + draft
                                                                                                                                                                                                                      
FRP_PIPELINE PACKAGE BODY  176     l_parsed  := parse_rfp(p_object_name);
                                                                                                                                                                                                           
FRP_PIPELINE PACKAGE BODY  177     l_matches := match_requirements(l_parsed, 5);
                                                                                                                                                                                                    
FRP_PIPELINE PACKAGE BODY  178     l_draft   := draft_proposal(l_parsed, l_matches);
                                                                                                                                                                                                
FRP_PIPELINE PACKAGE BODY  179 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  180     -- 3. Pull headline fields out of parsed JSON
                                                                                                                                                                                                    
FRP_PIPELINE PACKAGE BODY  181     BEGIN l_client := JSON_VALUE(l_parsed, '$.client_name'); EXCEPTION WHEN OTHERS THEN NULL; END;
                                                                                                                                                   
FRP_PIPELINE PACKAGE BODY  182     BEGIN l_due    := JSON_VALUE(l_parsed, '$.due_date');    EXCEPTION WHEN OTHERS THEN NULL; END;
                                                                                                                                                   
FRP_PIPELINE PACKAGE BODY  183 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  184     -- 4. Persist
                                                                                                                                                                                                                                    
FRP_PIPELINE PACKAGE BODY  185     INSERT INTO FRP_PROPOSALS (proposal_id, rfp_doc_id, client_name, due_date,
                                                                                                                                                                       
FRP_PIPELINE PACKAGE BODY  186                                proposal_status, extracted_json, draft_text)
                                                                                                                                                                          
FRP_PIPELINE PACKAGE BODY  187     VALUES (l_proposal_id, l_rfp_doc_id, l_client,
                                                                                                                                                                                                   
FRP_PIPELINE PACKAGE BODY  188             CASE WHEN l_due IS NOT NULL THEN TO_DATE(l_due,'YYYY-MM-DD') END,
                                                                                                                                                                        
FRP_PIPELINE PACKAGE BODY  189             'draft', l_parsed, l_draft);
                                                                                                                                                                                                             
FRP_PIPELINE PACKAGE BODY  190 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  191     -- 5. Write the draft back to the bucket as Markdown
                                                                                                                                                                                             
FRP_PIPELINE PACKAGE BODY  192     DBMS_CLOUD.PUT_OBJECT(
                                                                                                                                                                                                                           
FRP_PIPELINE PACKAGE BODY  193       credential_name => 'OCI$RESOURCE_PRINCIPAL',
                                                                                                                                                                                                   
FRP_PIPELINE PACKAGE BODY  194       object_uri      => FRP_INGEST.C_BUCKET_BASE || '03_IN_PROGRESS/' || l_proposal_id || '.md',
                                                                                                                                                    
FRP_PIPELINE PACKAGE BODY  195       contents        => UTL_RAW.CAST_TO_RAW(DBMS_LOB.SUBSTR(l_draft, 32000, 1))
                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  196     );
                                                                                                                                                                                                                                               
FRP_PIPELINE PACKAGE BODY  197 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  198     COMMIT;
                                                                                                                                                                                                                                          
FRP_PIPELINE PACKAGE BODY  199     RETURN l_proposal_id;
                                                                                                                                                                                                                            
FRP_PIPELINE PACKAGE BODY  200   END process_dropzone;
                                                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY  201 
                                                                                                                                                                                                                                                     
FRP_PIPELINE PACKAGE BODY  202   -- ─────────────────────────────────────────────────────────
                                                                                                                                                                                       
FRP_PIPELINE PACKAGE BODY  203   -- search_library · ad-hoc vector search (returns ref cursor for APEX)
                                                                                                                                                                             
FRP_PIPELINE PACKAGE BODY  204   -- ─────────────────────────────────────────────────────────
                                                                                                                                                                                       
FRP_PIPELINE PACKAGE BODY  205   FUNCTION search_library(p_query IN VARCHAR2, p_top_k IN NUMBER DEFAULT 8,
                                                                                                                                                                          
FRP_PIPELINE PACKAGE BODY  206                           p_deal_status IN VARCHAR2 DEFAULT NULL)
                                                                                                                                                                                    
FRP_PIPELINE PACKAGE BODY  207     RETURN SYS_REFCURSOR IS
                                                                                                                                                                                                                          
FRP_PIPELINE PACKAGE BODY  208     l_cur  SYS_REFCURSOR;
                                                                                                                                                                                                                            
FRP_PIPELINE PACKAGE BODY  209     l_qvec VECTOR;
                                                                                                                                                                                                                                   
FRP_PIPELINE PACKAGE BODY  210   BEGIN
                                                                                                                                                                                                                                              
FRP_PIPELINE PACKAGE BODY  211     l_qvec := DBMS_VECTOR_CHAIN.UTL_TO_EMBEDDING(
